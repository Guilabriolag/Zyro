  function updateCar(dt) {
    const t = clamp(steer + ((keys.ArrowRight || keys.KeyD ? 1 : 0) - (keys.ArrowLeft || keys.KeyA ? 1 : 0)), -1, 1);
    const accel = gas || keys.ArrowUp || keys.KeyW;

    else if (stop) { if (V.speed > 0) V.speed = Math.max(-9, V.speed - 30 * dt); else V.speed = Math.max(-9, V.speed - 10 * dt); }
    else V.speed *= Math.exp(-.9 * dt);

    V.yaw += smoothSteer * (.82 / (1 + Math.abs(V.speed) * .055)) * sf * dt * Math.sign(V.speed || 1);

  // Jogador a pé — câmera RELATIVA ao personagem com pinch
  function updateWalk(dt) {
    let ix = walkInput.x, iz = walkInput.z;
    if (keys.KeyW || keys.ArrowUp) iz -= 1;
    if (keys.KeyS || keys.ArrowDown) iz += 1;
    if (keys.KeyA || keys.ArrowLeft) ix -= 1;
    if (keys.KeyD || keys.ArrowRight) ix += 1;
    const len = Math.hypot(ix, iz);
    let moving = false;
    if (len > .12) {
      moving = true; ix /= len; iz /= len;
      // direção do mundo relativa à câmera (camYaw)
      const fx = Math.sin(walkCamYaw), fz = Math.cos(walkCamYaw);
      const rx = Math.cos(walkCamYaw), rz = -Math.sin(walkCamYaw);
      const mx = rx * ix + fx * (-iz);
      const mz = rz * ix + fz * (-iz);
      WP.x += mx * 7 * dt; WP.z += mz * 7 * dt;
      playerYaw = Math.atan2(mx, mz);
      player.rotation.y = playerYaw;
    }
    if (cityRoll > 0) {
      cityRoll -= dt;
      const k = 4.5;
      WP.x += cityRollDir[0] * k * dt;
      WP.z += cityRollDir[1] * k * dt;
      player.rotation.x = (0.6 - cityRoll) / 0.6 * Math.PI * 2;
      if (cityRoll <= 0) player.rotation.x = 0;
    }
    WP.x = clamp(WP.x, -290, 290); WP.z = clamp(WP.z, -300, 520);
    player.position.set(WP.x, 0, WP.z);
    if (moving) walkAnimTime += dt * 9; else walkAnimTime += dt * 1.5;
    const s = moving ? Math.sin(walkAnimTime) * .6 : 0;
    player.userData.legL.rotation.x = s;
    player.userData.legR.rotation.x = -s;
    player.userData.armL.rotation.x = -s * .7;
    player.userData.armR.rotation.x = s * .7;
    player.position.y = moving ? Math.abs(Math.sin(walkAnimTime)) * .05 : 0;
    // câmera orbitando o personagem
    const hd = walkCamDist * Math.cos(walkCamPitch);
    const vd = walkCamDist * Math.sin(walkCamPitch);
    const tx = WP.x + Math.sin(walkCamYaw + Math.PI) * hd;
    const tz = WP.z + Math.cos(walkCamYaw + Math.PI) * hd;
    const ty = 1.5 + vd;
    camera.position.lerp(new THREE.Vector3(tx, ty, tz), 1 - Math.exp(-dt * 8));
    camera.lookAt(WP.x, 1.35, WP.z);
  }
